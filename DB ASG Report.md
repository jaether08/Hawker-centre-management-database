DROP TABLE IF EXISTS RentalAgreement;

DROP TABLE IF EXISTS MenuItemCuisine;

DROP TABLE IF EXISTS Likes;

DROP TABLE IF EXISTS Complaint;

DROP TABLE IF EXISTS Feedback;

DROP TABLE IF EXISTS InspectionRemark;

DROP TABLE IF EXISTS Inspection;

DROP TABLE IF EXISTS OrderItem;

DROP TABLE IF EXISTS CustOrder;

DROP TABLE IF EXISTS MenuItem;

DROP TABLE IF EXISTS FoodStall;

DROP TABLE IF EXISTS HawkerCentre;

DROP TABLE IF EXISTS NEA_Officer;

DROP TABLE IF EXISTS Cuisine;

DROP TABLE IF EXISTS StallOwner;

DROP TABLE IF EXISTS Customer;

DROP TABLE IF EXISTS Operator;

CREATE TABLE Operator (

OperatorID CHAR(5) NOT NULL, -- OP\*\*\*

OperatorName VARCHAR(255) NOT NULL,

ContactPerson CHAR(10) NOT NULL,

CONSTRAINT PK_Operator PRIMARY KEY (OperatorID)

)

CREATE TABLE Customer (

CustomerID CHAR(5) NOT NULL, -- CS\*\*\*

CustNRIC VARCHAR(9) NOT NULL UNIQUE,

CustName VARCHAR(100) NOT NULL,

CustContactNo CHAR(10) NOT NULL UNIQUE,

CustEmail VARCHAR(50) NULL UNIQUE,

CONSTRAINT PK_CustomerID PRIMARY KEY (CustomerID)

);

CREATE TABLE StallOwner (

OwnerID CHAR(5) NOT NULL, -- OW\*\*\*

OwnerName VARCHAR(100) NOT NULL,

OwnerNRIC CHAR(9) NOT NULL UNIQUE,

OwnerContactNo VARCHAR(15) NOT NULL,

CONSTRAINT PK_OwnerIdentifier PRIMARY KEY (OwnerID)

);

CREATE TABLE Cuisine (

CuisineID CHAR(5) NOT NULL, --CU\*\*\*

CuisineDesc VARCHAR(50) NOT NULL,

CONSTRAINT PK_Cuisine PRIMARY KEY (CuisineID)

);

CREATE TABLE NEA_Officer (

OfficerID CHAR(5) NOT NULL, --OC\*\*\*

OfficerName VARCHAR(255) NOT NULL,

OfficerContactNo CHAR(10) NOT NULL,

CONSTRAINT PK_Officer PRIMARY KEY (OfficerID),

)

CREATE TABLE HawkerCentre (

HawkerCentreID CHAR(5) NOT NULL, --HC\*\*\*

HCName VARCHAR(100) NOT NULL,

HCAddress VARCHAR(200) NOT NULL,

OperatorID CHAR(5) NOT NULL,

CONSTRAINT PK_HCentreID PRIMARY KEY (HawkerCentreID),

CONSTRAINT FK_Operator_HCentre FOREIGN KEY (OperatorID) REFERENCES
Operator(OperatorID)

);

CREATE TABLE FoodStall (

StallID CHAR(5) NOT NULL, --ST\*\*\*

StallUnitNo VARCHAR(10) NOT NULL,

StallName VARCHAR(100) NOT NULL,

StallDesc VARCHAR(200),

HawkerCentreID CHAR(5) NOT NULL,

CONSTRAINT PK_StallIdentifier PRIMARY KEY (StallID),

CONSTRAINT FK_FoodStall_HCentre FOREIGN KEY (HawkerCentreID) REFERENCES
HawkerCentre(HawkerCentreID)

);

CREATE TABLE MenuItem (

StallID CHAR(5) NOT NULL,

ItemCode CHAR(5) NOT NULL, --IC\*\*\*

ItemDesc VARCHAR(100) NOT NULL,

ItemPrice DECIMAL(10,2) NOT NULL CHECK (ItemPrice\>0),

ItemCategory VARCHAR(50) NULL,

CONSTRAINT PK_MenuItem PRIMARY KEY (StallID, ItemCode),

CONSTRAINT FK_MenuItem_Stall FOREIGN KEY (StallID) REFERENCES
FoodStall(StallID)

);

CREATE TABLE CustOrder (

OrderID CHAR(5) NOT NULL, --OD\*\*\*

OrderDate DATETIME NOT NULL,

PmtType VARCHAR(30) NOT NULL,

CustomerID CHAR(5) NOT NULL,

CONSTRAINT PK_CustOrder PRIMARY KEY (OrderID),

CONSTRAINT FK_CustOrder_Customer FOREIGN KEY (CustomerID) REFERENCES
Customer(CustomerID),

CONSTRAINT CK_CustOrder CHECK (OrderDate \<= GETDATE())

);

CREATE TABLE OrderItem (

OrderID CHAR(5) NOT NULL,

OrderItemNo TINYINT NOT NULL,

StallID CHAR(5) NOT NULL,

ItemCode CHAR(5) NOT NULL,

Quantity TINYINT NOT NULL,

UnitPrice DECIMAL(10,2) NOT NULL CHECK (UnitPrice \> 0),

CONSTRAINT PK_OrderItem PRIMARY KEY (OrderID, OrderItemNo),

CONSTRAINT FK_OrderItem_Order FOREIGN KEY (OrderID) REFERENCES
CustOrder(OrderID),

CONSTRAINT FK_OrderItem_MenuItem FOREIGN KEY (StallID, ItemCode)
REFERENCES MenuItem(StallID, ItemCode)

);

CREATE TABLE Inspection (

InspectionID CHAR(5) NOT NULL, --IN\*\*\*

InspectionDate DATETIME NOT NULL,

HygieneGrade CHAR(1) NOT NULL,

GradeExpiry DATETIME NOT NULL,

OfficerID CHAR(5) NOT NULL,

StallID CHAR(5) NOT NULL,

CONSTRAINT PK_Inspection PRIMARY KEY (InspectionID),

CONSTRAINT FK_Inspection_Officer FOREIGN KEY (OfficerID) REFERENCES
NEA_Officer(OfficerID),

CONSTRAINT FK_Inspection_Stall FOREIGN KEY (StallID) REFERENCES
FoodStall(StallID),

CONSTRAINT InspectionDate CHECK (InspectionDate \<= GETDATE()),

CONSTRAINT HygieneGrade CHECK (HygieneGrade IN ('A','B','C'))

)

CREATE TABLE InspectionRemark (

InspectionID CHAR(5) NOT NULL,

InspectionRemark VARCHAR(700) NOT NULL,

CONSTRAINT PK_InspectionRemark PRIMARY KEY (InspectionID,
InspectionRemark),

CONSTRAINT FK_InspectionRemark_Inspection FOREIGN KEY (InspectionID)
REFERENCES Inspection(InspectionID)

)

CREATE TABLE Feedback (

FbkID CHAR(5) NOT NULL, --FB\*\*\*

FbkComment VARCHAR(500),

FbkDateTime DATETIME NOT NULL,

FbkRating TINYINT NOT NULL CHECK (FbkRating IN (1,2,3,4,5)),

CustomerID CHAR(5) NOT NULL,

StallID CHAR(5) NOT NULL,

CONSTRAINT PK_FbkID PRIMARY KEY (FbkID),

CONSTRAINT FK_Feedback_CustomerID FOREIGN KEY (CustomerID) REFERENCES
Customer(CustomerID),

CONSTRAINT FK_Feedback_StallID FOREIGN KEY (StallID) REFERENCES
FoodStall(StallID)

);

CREATE TABLE Complaint (

FbkID CHAR(5) NOT NULL,

Category CHAR(10) NOT NULL,

CONSTRAINT PK_ComplaintID PRIMARY KEY (FbkID),

CONSTRAINT FK_FbkID FOREIGN KEY (FbkID) REFERENCES Feedback(FbkID),

);

CREATE TABLE Likes (

CustomerID CHAR(5) NOT NULL,

StallID CHAR(5) NOT NULL,

ItemCode CHAR(5) NOT NULL,

CONSTRAINT PK_Likes PRIMARY KEY (CustomerID, StallID, ItemCode),

CONSTRAINT FK_Likes_CustomerID FOREIGN KEY (CustomerID) REFERENCES
Customer(CustomerID),

CONSTRAINT FK_Likes_StallIDItemCode FOREIGN KEY (StallID, ItemCode)
REFERENCES MenuItem(StallID, ItemCode)

);

CREATE TABLE MenuItemCuisine (

CuisineID CHAR(5) NOT NULL,

StallID CHAR(5) NOT NULL,

ItemCode CHAR(5) NOT NULL,

CONSTRAINT PK_MenuItemCuisine PRIMARY KEY (CuisineID, StallID,
ItemCode),

CONSTRAINT FK_MIC_Cuisine FOREIGN KEY (CuisineID) REFERENCES
Cuisine(CuisineID),

CONSTRAINT FK_MIC_MenuItem FOREIGN KEY (StallID, ItemCode) REFERENCES
MenuItem(StallID, ItemCode)

);

CREATE TABLE RentalAgreement (

AgreementID CHAR(5) NOT NULL, --AG\*\*\*

AgrStartDate DATETIME NOT NULL,

AgrEndDate DATETIME NOT NULL,

AgrTermCondition VARCHAR(8000) NULL,

RentalPrice DECIMAL(10,2) NOT NULL,

OwnerID CHAR(5) NOT NULL,

StallID CHAR(5) NOT NULL,

CONSTRAINT PK_RentalAgreement PRIMARY KEY (AgreementID),

CONSTRAINT FK_Rental_Owner FOREIGN KEY (OwnerID) REFERENCES
StallOwner(OwnerID),

CONSTRAINT FK_Rental_Stall FOREIGN KEY (StallID) REFERENCES
FoodStall(StallID)

);

CREATE TABLE Promotion (

PromoID CHAR(5) NOT NULL, --PR\*\*\*

PromoDesc VARCHAR(255) NOT NULL,

PromoStartDate DATETIME NOT NULL,

PromoEndDate DATETIME NOT NULL,

StallID CHAR(5) NOT NULL,

CONSTRAINT PK_Promotion PRIMARY KEY (PromoID),

CONSTRAINT FK_Promotion_Stall FOREIGN KEY (StallID) REFERENCES
FoodStall(StallID)

);

### FoodStalls that has RAs expiring soon

The following SQL query is used to track the Rental Agreement of each
stall before they expire; this allows the business to quickly inform the
Owner to reapply for another Rental Agreement when approaching the due
date.

#### English Query

List the HawkerCentreID, Hawker Centre Name, StallID, StallName,
StallUnitNo, Agreement End Date and Months until the Agreement expires,
calculated by subtracting the Month of the Agreement End Date and the
Month of the Current Date. With the number of months being ordered in
ascending order.

#### SQL Query

SELECT hc.HawkerCentreID, hc.HCName AS 'Hawker Centre Name', fs.StallID,
StallName, StallUnitNo, AgrEndDate,

DATEDIFF(MONTH,GETDATE(), AgrEndDate) AS 'Months till expired' FROM
FoodStall fs INNER JOIN HawkerCentre hc

ON fs.HawkerCentreID = hc.HawkerCentreID

INNER JOIN RentalAgreement ra ON fs.StallID = ra.StallID

ORDER BY \[Months till expired\] ASC, hc.HawkerCentreID ASC, fs.StallID
ASC

### Total Sales Revenue by Payment Type

The query below analyzes customers’ preferred payment methods and their
contribution to the overall revenue. This helps operators understand
which payment channels are the most used among paying customers. From a
business point of view, since cash is the most popular, stall owners can
ensure their staff are trained to handle cash and have sufficient change
for peak periods. Since Credit Card users spend the most among all
payment types, they can create premium menu items to target Credit Card
users and earn more money. This also directly leads to the provision of
faster service and better hygiene, since stall staffs would not have to
touch the cash. Operators can also implement incentives to encourage
payment through these cashless channels.

#### English Query

List payment methods used by customers, total number of orders, total
items ordered, total revenue generated, and average item value for each
payment method. Sorted in descending order by total revenue.

count(distinct co.OrderID) as "TotalOrders",

round(avg(oi.Quantity\*oi.UnitPrice), 2) as "AvgItemValue"

from CustOrder co inner join OrderItem oi on co.OrderID = oi.OrderID

group by co.PmtType

order by “TotalRevenue” desc;

### Operators with best Stalls

This compares all the operators to see who’s doing the best job
maintaining the hygiene standards. This is so to identify those stalls

#### English Query

Compare all the hawker centre operators and tell me how many centres
each one manages and how many of their stalls have Grade A hygiene
ratings. I want to see which operators are maintaining the best hygiene
standards across their centres.

#### SQL Query

SELECT hc.HawkerCentreID, hc.HCName, o.OperatorName,

(SELECT COUNT(DISTINCT fs1.StallID) FROM FoodStall fs1

INNER JOIN HawkerCentre hc1 ON fs1.HawkerCentreID = hc1.HawkerCentreID
